### Download workbook, extract data, and update JSONModel

Source: https://docs.sheetjs.com/docs/demos/frontend/openui5/

Fetches a workbook from the SheetJS docs URL, parses it with XLSX.read, converts the first worksheet to an array of objects with sheet_to_json, and updates the JSONModel's '/presidents' property.

```JavaScript
_loadExcelFile: async function () {
  /* Download from https://docs.sheetjs.com/pres.xlsx */
  const f = await (await fetch("https://docs.sheetjs.com/pres.xlsx")).arrayBuffer();

  /* parse */
  const wb = XLSX.read(f); // parse the array buffer
  /* generate array of objects from first worksheet */
  const ws = wb.Sheets[wb.SheetNames[0]]; // get the first worksheet
  const data = XLSX.utils.sheet_to_json(ws); // generate objects

  /* update JSONModel */
  this.getView().getModel().setProperty("/presidents", data);
}
```

### React data grid example with react-data-grid and xlsx

Source: https://docs.sheetjs.com/docs/solutions/output/

Fetches a remote Excel file, reads it with XLSX.read, and uses sheet_to_json with header: 1 to generate rows and columns for react-data-grid. Requires react-data-grid and xlsx packages.

```javascript
import { useEffect, useState } from "react";
import DataGrid from "react-data-grid";
import { read, utils } from "xlsx";

import 'react-data-grid/lib/styles.css';

const url = "https://docs.sheetjs.com/pres.xlsx";

export default function App() {
  const [columns, setColumns] = useState([]);
  const [rows, setRows] = useState([]);
  useEffect(() => {(async () => {
    const wb = read(await (await fetch(url)).arrayBuffer());

    /* use sheet_to_json with header: 1 to generate an array of arrays */
    const data = utils.sheet_to_json(wb.Sheets[wb.SheetNames[0]], { header: 1 });

    /* see react-data-grid docs to understand the shape of the expected data */
    setColumns(data[0].map((r) => ({ key: r, name: r })));
    setRows(data.slice(1).map((r) => r.reduce((acc, x, i) => {
      acc[data[0][i]] = x;
      return acc;
    }, {})));
  })(); });

  return <DataGrid columns={columns} rows={rows} />;
}
```
