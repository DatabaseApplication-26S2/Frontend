### Import and render the Calendar component

Source: https://github.com/shadcn-ui/ui/blob/shadcn@4.21.0/apps/v4/content/docs/components/aria/calendar.mdx

Import required date utilities and the Calendar component, manage selected date state, and render the calendar.

```tsx
import {
  getLocalTimeZone,
  today,
  type CalendarDate,
} from "@internationalized/date"

import { Calendar } from "@/components/ui/calendar"
```

```tsx
const [date, setDate] = React.useState<CalendarDate | null>(
  today(getLocalTimeZone())
)

return (
  <Calendar value={date} onChange={setDate} className="rounded-lg border" />
)
```
